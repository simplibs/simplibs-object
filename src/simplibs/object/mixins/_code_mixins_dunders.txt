class CollectionBase(SimpleMixin):
    """
    Base mapping mixin for composite instances.
    Builds a frozen mapping {name_snake: inner_instance} and provides
    the basic keys(), values(), items() API.
    """

    # ----------------------------------------
    # Basic interface
    # ----------------------------------------

    # Compatibility declaration
    _composite_only = True

    # Slot definition
    __slots__ = ("_inners_name_set", "_inners_name_tuple")

    # ----------------------------------------
    # Initialization hook
    # ----------------------------------------

    # noinspection PyAttributeOutsideInit
    def init_mixins(
        self: "CollectionBaseProtocol"
    ) -> None:
        """
        Initializes two immutable structures for working with inners:

        - `_inners_name_tuple`: tuple of all name_snake keys in the
          deterministic order defined by the blueprint. Serves as the
          source for iteration, reversed(), keys(), values() and items().

        - `_inners_name_set`: set of all name_snake keys. Used for fast
          existence checks (O(1) lookup).

        Both structures are derived only from direct inners and are
        immutable for the entire lifetime of the instance.
        """

        # 1. Initialize parents
        super().init_mixins()

        # 2. Tuple holding name_snake in exact order
        self._inners_name_tuple: "NamesTuple" = tuple(
            inner.meta.name_snake
            for inner in self.meta.inners
        )

        # 3. Set for fast presence comparison
        self._inners_name_set: "NamesSet" = set(self._inners_name_tuple)

    # ----------------------------------------
    # Methods - KEYS / VALUES / ITEMS
    # ----------------------------------------

    def keys(
        self: "CollectionBaseProtocol"
    ) -> list[str]:
        """Returns a list of keys (name_snake) of all direct inners."""
        return list(self._inners_name_tuple)

    def values(
        self: "CollectionBaseProtocol"
    ) -> list[object]:
        """Returns a list of values corresponding to each inner."""
        return [
            getattr(self, key)
            for key in self._inners_name_tuple
        ]

    def items(
        self: "CollectionBaseProtocol"
    ) -> list[tuple[str, object]]:
        """Returns a list of (key, value) pairs for all inners."""
        return [
            (key, getattr(self, key))
            for key in self._inners_name_tuple
        ]

    Provides access via obj[key], value assignment, and existence checks.
    """

    # ----------------------------------------
    # Basic interface
    # ----------------------------------------

    # Compatibility declaration
    _composite_only = True

    # Slot definition
    __slots__ = ()

    # ----------------------------------------
    # Methods - LEN / CONTAINS / MISSING
    # ----------------------------------------

    def __len__(
        self: "MappingProtocol"
    ) -> int:
        """Returns the number of direct inners."""
        return len(self._inners_name_tuple)

    def __contains__(
        self: "MappingProtocol",
        key: str
    ) -> bool:
        """Returns True if the composite contains an inner with the given key."""
        return key in self._inners_name_set

    def __missing__(
        self: "MappingProtocol",
        key: str
    ) -> None:
        """Raises KeyError with information about the available keys."""
        raise KeyError(
            f"Item '{key}' is not defined on {self.meta.class_name}. "
            f"Available items: {self.keys()}"
        )

    # ----------------------------------------
    # Methods - GETITEM / GET / SETITEM
    # ----------------------------------------

    def __getitem__(
        self: "MappingProtocol",
        key: str
    ):
        """Returns the value of an inner by key, or raises via __missing__."""
        if key not in self._inners_name_set:
            self.__missing__(key)
        return getattr(self, key)

    def get(
        self: "MappingProtocol",
        key: str,
        default: Any = None
    ):
        """Safely returns the value of an inner, or default if the key doesn't exist."""
        if key not in self._inners_name_set:
            return default
        return getattr(self, key)

    def __setitem__(
        self: "MappingProtocol",
        key: str,
        value: Any
    ):
        """Sets the value of an inner by key, if it exists."""

        # 1. Check that we are not on an immutable object
        if not self.__is_mutable__:
            raise TypeError(
                f"Immutable composite '{self.meta.class_name}' "
                f"does not allow changing values via __setitem__."
            )

        # 2. Check that the key exists
        if key not in self._inners_name_set:
            self.__missing__(key)

        # 3. Request the value change
        # (May be blocked on a composite without decomposition - an
        # exception is raised in that case)
        setattr(self, key, value)

class Equality(SimpleMixin):
    """
    Equality and hashing for Simple instances.

    Equality (__eq__) is value-based:
    - If the other operand is a Simple instance of the same blueprint,
      their values are compared.
    - If the other operand is a raw value, it is compared directly.

    Hashing (__hash__) is driven by instance mutability:
    - For mutable instances (__is_mutable__ = True), identity-based hash
      (id(self)) is used, to guarantee stability even when the value
      changes.
    - For immutable instances (__is_mutable__ = False), value-based hash
      (hash(self.value)) is used, because the value is guaranteed to be
      immutable and hashable.

    This mixin is universal — it can be used for both mutable and
    immutable instances, since the hashing strategy is chosen dynamically
    based on the __is_mutable__ flag.
    """

    # ----------------------------------------
    # Basic interface
    # ----------------------------------------

    # Slot definition
    __slots__ = ()

    # ----------------------------------------
    # Methods - EQ / HASH
    # ----------------------------------------

    def __eq__(
        self: "EqualityProtocol",
        other: Any
    ) -> bool:
        """
        Value-based equality.
        - If the other operand is a Simple instance of the same
          blueprint, their values are compared.
        - If the other operand is a raw value, it is compared directly.
        """
        # 1. Comparing two Simple instances
        if hasattr(other, "meta") and hasattr(other, "value"):
            if self.meta.blueprint_path == other.meta.blueprint_path:
                return self.value == other.value
            return False

        # 2. Comparing against a raw value
        return self.value == other

    def __hash__(
        self: "EqualityProtocol"
    ) -> int:
       """
       Returns a hash based on the instance's mutability.

       - For mutable instances (__is_mutable__ = True), returns a hash
         based on object identity (id(self)), to guarantee stability even
         when the value changes.

       - For immutable instances (__is_mutable__ = False), returns a
         value-based hash (hash(self.value)), because the value is
         guaranteed to be immutable and hashable.

       This method thus unifies hashing for both mutable and immutable
       mode within a single mixin.
       """
       return (
           id(self)
           if self.__is_mutable__ else
           hash(self.value)
       )

class ClassGetItem(SimpleMixin):
    """
    Mixin providing support for generic-style subscript syntax:

        MyClass[int]
        MyClass[str]
        MyClass[tuple[int, str]]

    This method is purely a syntactic aid for type hints, IDEs, and
    static analysis. It has no effect on the instance's runtime behavior
    or on the construction of the instance class.

    Logic:
    - If a parent class already implements __class_getitem__, that
      implementation is used.
    - If it doesn't exist (the common case for plain classes), the class
      itself is returned.
    """

    # ----------------------------------------
    # Basic interface
    # ----------------------------------------

    # Slot definition
    __slots__ = ()

    # ----------------------------------------
    # Classmethods - CLASS GETITEM
    # ----------------------------------------

    @classmethod
    def __class_getitem__(
        cls: type[Self],
        item: Any
    ) -> type[Self] | Any:
        """
        Enables the MyClass[T] syntax without having to inherit from
        typing.Generic.

        If the parent class implements __class_getitem__, its
        implementation is used. Otherwise the class itself is returned,
        which is the standard fallback Python uses for simple generic
        subscript syntax.
        """

        # 1. Look up the method on the parent class
        super_cls = super(ClassGetItem, cls)
        method = getattr(super_cls, "__class_getitem__", None)

        # 2. Return the method's result, or the class as a fallback
        return (
            method(item)
            if method is not None
            else cls
        )

        obj("a", 10)
        obj(("a", 10), ("b", 20))

    - All processing is delegated to the internal `with_any` engine.
    - Adds no validation or logic of its own.
    - Universal for both mutable and immutable objects.
    """

    # Basic interface
    # ----------------------------------------

    # Slot definition
    __slots__ = ()

    # ----------------------------------------
    # Methods - CALL
    # ----------------------------------------

    def __call__(
        self: "CallProtocol",
        *changes: Any,
        **kwargs: Any
    ) -> "CallProtocol":
        """
        Allows the instance to be called like a function.

        Supported forms:
            obj("a", 10)
            obj(("a", 10))
            obj(("a", 10), ("b", 20))
            obj((("a", 10), ("b", 20)))

        If keyword arguments are used, they are converted to tuple form.

        # 1. If keyword arguments are used → convert them to tuple form
        if kwargs:
            # kwargs always take the form of constants
            changes = tuple(kwargs.items())

        elif len(changes) == 1 and isinstance(changes[0], tuple):
            changes = changes[0]

        # 3. Delegate to the engine
        return with_any(self, changes)

    This mixin implements the __getstate__ and __setstate__ dunders so
    that serialization happens through a readable dict (to_dict) and
    deserialization through the from_dict method. This keeps pickling
    stable, readable, and independent of the object's internal
    representation.

    Requirements:
    - The class MUST provide a from_dict() method (from
      MutableSerialization or ImmutableSerialization).
    - The mixin works both for mutable instances (in-place mutation) and
      immutable instances (from_dict returns a new instance, which pickle
      automatically uses).

        Pickle stores this dict as the object's representation. The
        benefit is that serialization is stable, readable, and
        independent of the instance's internal and private attributes.
        """
        return self.to_dict()

    def __setstate__(
        self: "PickleStateProtocol",
        state: "StateDict"

        raises TypeError with a clear explanation. For mutable instances,

                f"(provided by MutableSerialization or ImmutableSerialization)."

def _align_operand_type(
    self: "SimpleMixinLike",
    value: Any
) -> Any:
    """
    Attempts to convert the operand to the default value type,
    but only when the conversion is safe.

    Safe conversions:
    - int → float
    - int → Decimal
    - int → Fraction

    Unsafe conversions (NOT performed):
    - float → int
    - float → Decimal (rounding issues)
    - Fraction → int
    - Decimal → int
    """

    # 1. Determine target type
    target_type = type(self.meta.default)

    # 2. Attempt safe conversion
    try:
        return target_type(value)

    # 3. On failure, return original value
    except Exception:
        return value

def execute_inplace_op(

    operator: "Operator",
    other: Any

    Fast modular executor for in‑place arithmetic operations.
    Extracts the operand, performs the computation, and safely writes the
    result into self.value. Includes fallback for safe type alignment.
    """

    # 1. Extract second operand
    other_value = getattr(other, "value", other)

    # 2. Attempt direct computation
    try:
        self.value = operator(self.value, other_value)
        return self

    # 3. Attempt safe type alignment
    except TypeError:

        # 3.1 Align operand type
        coerced = _align_operand_type(self, other_value)

        try:
            self.value = operator(self.value, coerced)
            return self

        # 3.3 Failure or division by zero
        except (TypeError, ZeroDivisionError):
            return NotImplemented

    # 3.3 Division by zero in the first attempt
    except ZeroDivisionError:
        return NotImplemented

        self: "ValueDisplayProtocol",

        If no pattern is given, uses the mask from metadata (meta.mask),

        composite values.
        """
        value = self.value
        mask = pattern or getattr(self.meta, "mask", None)

        if mask and isinstance(value, (str, int, float)):

    ) -> str:
