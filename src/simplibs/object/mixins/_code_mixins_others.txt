class ImmutableMethods(Infrastructure):
    """
    Mixin providing the immutable API for working with instances:

    - with_constants(...)
    - with_constant(...)
    - with_composite(...)

    Construction of immutable classes (getter-only value, removal of
    setters, removal of mutating methods, immutable inners) is now
    handled directly in make_atom_class / make_composite_class.
    """

    # ----------------------------------------
    # Basic interface
    # ----------------------------------------

    # Compatibility declaration
    _immutable_only = True

    # Slot definition
    __slots__ = ()

    # ----------------------------------------
    # Methods - Immutable API (core)
    # ----------------------------------------

    def with_constants(
        self: "ImmutableMethodsProtocol",
        changes_tuple: "ChangesTuple | None" = None
    ) -> "ImmutableMethodsProtocol":
        """Creates a new immutable instance based on multiple changes."""
        return with_constants(self, changes_tuple)

    def with_constant(
        self: "ImmutableMethodsProtocol",
        name: str,
        new_value: Any
    ) -> "ImmutableMethodsProtocol":
        """Creates a new immutable instance by changing a single constant."""
        return with_constant(self, name, new_value)

    def with_composite(
        self: "ImmutableMethodsProtocol",
        name: str,
        new_value: Any
    ) -> "ImmutableMethodsProtocol":
        """Creates a new immutable instance by changing a single composite."""
        return with_composite(self, name, new_value)

    # Methods - Immutable API (extended)
    # ----------------------------------------

        self: "ImmutableMethodsProtocol",
        changes_dict: "ChangesDict"

class ImmutableWithInners(ImmutableMethods):
    """
    Adds, on top of the base class's basic methods, dedicated methods for
    every inner.

    Guarantees the presence of dynamically generated immutable methods
    (setters) for each inner element (e.g. `obj.with_uzel(value)`).
    Every one of these methods produces a new, transformed instance (Self).
    """

    # ----------------------------------------
    # Basic interface
    # ----------------------------------------

    # Compatibility declaration
    _immutable_only = True
    _composite_only = True

    # Slot definition
    __slots__ = ()

    # ----------------------------------------
    # Hook for modifying the instance class
    # ----------------------------------------

    @staticmethod
    def __extend_instance_class__(
        instance_cls: "SimpleInstanceLike"
    ) -> "SimpleInstanceLike":
        """
        Internal method for attaching elements to the class, before the
        initialization process.
        """

        # 1. Iterate over nodes
        for name in instance_cls.meta.node_names_set:

            # 2. Method definition
            # (All use the composite method, since it forwards constants to with_constant)
            def method(self: "ImmutableMethodsWithInnersProtocol", value: Any, _name: str=name):
                return self.with_composite(_name, value)

            # 3. Set the inner element's attributes
            method_name = f"with_{name}"
            method.__name__ = method_name
            method.__doc__ = (
                f"Immutable setter for the value `{name}`.\n"
                f"Returns a new immutable instance with `{name}` changed."
            )

            # 4. Create the method attribute
            setattr(instance_cls, method_name, method)

        # 5. Return the modified class
        return instance_cls

def verify_down_limit(
    self: "StepperProtocol",
    new_value: Any,
    *,
    raise_exception: bool = False
) -> bool:
    """
    Verifies that the new value does not fall below the stepper's down limit.

    If the down limit is set and the new value is below it,
    triggers the optional limit action and optionally raises a structured error.
    """

    # 1. Check whether the DOWN limit is set and violated
    if self.down_limit is not UNSET and new_value < self.down_limit:

        # 2. Trigger limit action if present
        if self.down_limit_action:
            self.down_limit_action()

        # 3. Raise structured error if requested
        if raise_exception:
            raise_stepper_down_limit_reached(self, new_value)

        return False

    # 4. Successful validation
    return True

def set_value(
    self: "StepperProtocol",
    value: Any,
    *,
    validate: bool = True,
    raise_exception: bool = False
) -> Any:
    """
    Safely writes a new value into the instance with full limit and action control.
    """

    # 1. Normalize and validate the value
    _value = process_value(self, value, validate=validate)

    # 2. Handle identical value (no-op)
    # noinspection PyProtectedMember
    if _value == self._instance._value:
        return True

    # 3. Limit checks
    # noinspection PyProtectedMember
    if _value > self._instance._value:
        if not verify_up_limit(self, _value, raise_exception=raise_exception):
            return False
    else:
        if not verify_down_limit(self, _value, raise_exception=raise_exception):
            return False

    # 4. Write the value
    self._instance._value = _value
    return True
