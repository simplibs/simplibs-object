def get_top_composite_paths(
    composite: "ChangesTuple",
    nodes_path_dict: "NodesPathDict",
) -> tuple["PathDict", "PathDict"]:
    """
    Returns only the composite paths closest to the root — i.e., those that
    are not subpaths of any other composite path in the dictionary.

    Logic:
    - Sort paths by length (shorter = higher in hierarchy)
    - For each path, check whether it is a prefix of any already stored root path
    - If not a prefix → it is a new root composite
    """

    # 1. Prepare variables
    top_composite = {}
    top_composite_paths = []
    child_composite = {}

    # 2. Build sorted list of (name, path) pairs
    composite_names = (name for name, _ in composite)
    sorted_paths = sorted(
        ((name, nodes_path_dict[name]) for name in composite_names),
        key=lambda item: len(item[1]),
    )

    # 3. Filter root composite paths
    for key, path in sorted_paths:

        # 3.1 Process top-level composites
        if not any(
            path[:len(parent)] == parent
            for parent in top_composite_paths
        ):
            top_composite[key] = path
            top_composite_paths.append(path)

        # 3.2 Process inner composites
        else:
            child_composite[key] = path

    # 4. Return both dictionaries
    return top_composite, child_composite

def split_constants_paths(
    constants: "ChangesTuple",
    used_paths_dict: "PathDict",
    nodes_path_dict: "NodesPathDict",
) -> tuple["PathDict", "PathDict"]:
    """
    Splits constant paths into two groups:

    - contained: paths that lie inside one of the provided root paths
    - unique:    paths that do not fall under any root path
    """

    # 1. Build dictionary of constant paths
    constants_paths = {name: nodes_path_dict[name] for name, _ in constants}
    used_paths = used_paths_dict.values()

    # 2. Prepare containers
    contained = {}
    unique = {}

    # 3. Classify paths
    for key, path in constants_paths.items():

        # 3.1 Path starts with any root → contained
        if any(
            path[:len(used_path)] == used_path
            for used_path in used_paths
        ):
            contained[key] = path

        # 3.2 Otherwise → unique
        else:
            unique[key] = path

    # 4. Return both dictionaries
    return contained, unique

def with_any(
    self: "InfrastructureProtocol",
    changes: "ChangeTuple | ChangesTuple | None" = None,
) -> "InfrastructuredComponent | Any":
    """
    Integration requirement:
    This function is designed for the core pipeline framework. The object
    provided in `self` MUST inherit from:
      1. Snapshot (provides _restore_snapshot)
      2. NodesDicts (provides metadata and node access)
    Without these mixins, the pipeline will raise AttributeError.
    """

    # 1. Handle case with no changes
    if not changes:
        instance = create_new_instance(self, self._build_snapshot())
        return instance if not self.__is_mutable__ else instance.value

    # 2. Validate type of changes
    if not isinstance(changes, tuple):
        raise_with_any_changes_is_not_tuple(self, changes)

    # 3. Handle single change
    if not isinstance(changes[0], tuple):
        changes: tuple[str, Any]
        instance = process_single_change(self, changes)
        return instance if not self.__is_mutable__ else instance.value

    # 4. Split changes into constants and composites
    changes: tuple[tuple[str, Any], ...]
    constants, composite = split_changes(self, changes)

    # 5. If all changes are constants → delegate
    if not composite:
        instance = with_constants(self, changes)
        return instance if not self.__is_mutable__ else instance.value

    # 6. Build dictionary of provided values
    changes_dict = {key: value for key, value in changes}

    # 7. Determine top-level and child composite paths
    top_composite, child_composite = get_top_composite_paths(
        composite,
        self._nodes_path_dict,
    )

    # 8. Create base instance
    instance = create_new_instance(self, self._build_snapshot())

    # 9. Apply root composites
    for name in top_composite:
        instance = with_composite(instance, name, changes_dict[name])

    # 10. Snapshot after root composites
    # noinspection PyProtectedMember
    instance_snapshot_dict = create_snapshot_dict(instance._nodes_tuple)

    # 11. Validate child composites
    if child_composite:
        for name, value in composite:
            if instance_snapshot_dict[name] != value:
                raise_with_any_child_composite_mismatch(
                    self, name, value, instance_snapshot_dict[name]
                )

    # 12. Process constants
    if constants:

        # 12.1 Split into contained and unique constants
        contained, unique = split_constants_paths(
            constants,
            top_composite,
            self._nodes_path_dict,
        )

        # 12.2 Validate contained constants
        for name, _ in contained:
            if instance_snapshot_dict[name] != changes_dict[name]:
                raise_with_any_contained_constant_mismatch(
                    self, name, changes_dict[name], instance_snapshot_dict[name]
                )

        # 12.3 Apply unique constants
        new_changes = [(name, changes_dict[name]) for name, _ in unique]
        instance = with_constants(instance, tuple(new_changes))

    # 13. Return instance or value (mutable mode)
    return instance if not self.__is_mutable__ else instance.value

def reacalculate_inners(
    snapshot_dict: "SnapshotDict",
    meta: "SimpleMetadataLike",
    new_value: Any,
) -> None:
    """
    Recalculates the values of inner nodes for a composite based on the
    provided new composite value.
    """

    # 1. Load inner metadata objects
    inners_meta = (inner.meta for inner in meta.inners)

    # 2. Compute decomposed inner values
    inners_value = meta.decompose(new_value)

    # 3. Iterate through metadata and computed values
    for inner_meta, inner_value in zip(inners_meta, inners_value):

        # 3.1 Write the inner value
        snapshot_dict[inner_meta.name_snake] = inner_value

        # 3.2 If the inner node is itself a composite → recurse
        if not inner_meta.is_constant:
            reacalculate_inners(snapshot_dict, inner_meta, inner_value)

def find_longest_keys(
    path_dict: "PathDict",
) -> "NamesTuple":
    """
    Finds the keys whose paths represent terminal (deepest) nodes within
    the provided path dictionary.
    """

    # 1. Convert dictionary into a list of (key, path) pairs
    #    and sort them by their path (index 1)
    sorted_path = sorted(path_dict.items(), key=lambda x: x[1])
    longest_keys = []

    # 2. Iterate and compare paths
    for i in range(len(sorted_path) - 1):
        key, path = sorted_path[i]
        _, next_path = sorted_path[i + 1]

        # If the next path starts with the current path,
        # the current one is an intermediate node → skip it.
        if next_path[:len(path)] == path:
            continue

        # Otherwise, this is a terminal node → store its key.
        longest_keys.append(key)

    # The last element in the sorted list is always terminal
    longest_keys.append(sorted_path[-1][0])

    return tuple(longest_keys)

def verify_mixin_definition(
    cls: "SimpleMixinProtocol",
) -> None:
    """
    Performs structural validation of a mixin class definition.
    Ensures that the mixin is safe to merge into dynamically generated
    instance classes.
    """

    # 1. __slots__ must be explicitly defined
    if "__slots__" not in cls.__dict__:
        raise_mixin_slots_is_not_defined(cls)

    # 2. Extract slot definition
    slots = cls.__dict__["__slots__"]

    # 3. __slots__ must be tuple or string
    if not isinstance(slots, (tuple, str)):
        raise_mixin_slots_unsuport_type(cls)

    # 4. All slot items must be strings
    slots_set = set(slots) if isinstance(slots, tuple) else {slots}
    if not all(isinstance(s, str) for s in slots_set):
        raise_mixin_slots_contain_unvalid_type(cls)

    # 5. __slots__ must not contain forbidden names
    if FORBIDDEN_NAMES & slots_set:
        raise_mixin_slots_contains_forbidden_slot_names(cls, slots_set)

    # 6. Class-level namespace must not contain forbidden names
    if FORBIDDEN_NAMES & cls.__dict__.keys():
        raise_mixin_dict_contains_forbidden_attributes(cls)

def from_dict(
    self: "SerializationProtocol",
    user_dict: "StateDict",
) -> "SerializationProtocol":
    """
    Universal reconstruction of instance state from a dictionary.
    Supports both mutable (in-place) and immutable (new instance) modes.
    """

    # 1. Validate input type
    if not isinstance(user_dict, dict):
        raise_unvalid_state_dict_input(self, user_dict)

    # 2. Validate owner class
    owner_class = user_dict.get(OWNER_CLASS, UNSET)
    if self.meta.owner_class != owner_class:
        raise_unvalid_owner_class_mismatch_for_state_dict(self, owner_class)

    # 3. Prepare traversal structures
    values = []
    last_composite_key = UNSET
    last_composite_dict = {}

    # 4. Load and validate values according to the _nodes_tuple structure
    for instance, path in self._nodes_tuple:

        # 4.1 Define per-node variables
        current_dict = user_dict
        last_key = path[-1]

        # 4.2 Handle composite nodes (non-root)
        if len(path) > 1:

            # 4.2.1 Extract second-to-last key
            before_last_key = path[-2]

            # 4.2.2 Reuse previously cached composite dictionary
            if last_composite_key == before_last_key:
                current_dict = last_composite_dict

            # 4.2.3 Traverse into a new composite dictionary
            else:

                # 4.2.3.1 Traverse into nested dictionaries
                for key in path[:-1]:
                    current_dict = current_dict.get(key, UNSET)
                    if current_dict is UNSET:
                        raise_missing_composite_key(self, key, path)
                    if not isinstance(current_dict, dict):
                        raise_invalid_composite_structure(self, key, current_dict, path)

                # 4.2.3.2 Cache composite dictionary for next iteration
                if last_key == VALUE:
                    last_composite_key = before_last_key
                    last_composite_dict = current_dict

        # 4.3 Load value
        value = current_dict.get(last_key, UNSET)
        if value is UNSET:
            raise_missing_value_key(self, last_key, path)

        # 4.4 Validate and store value
        instance.validate(value)
        values.append(value)

    # 5. Mutable mode → write values in-place
    if self.__is_mutable__:
        for (instance, _), value in zip(self._nodes_tuple, values):
            instance._value = value
        return self

    # 6. Immutable mode → create new instance from snapshot
    new_snapshot = tuple(values)
    return create_new_instance(self, new_snapshot)

def from_dict_constants(
    self: "SerializationProtocol",
    user_dict: "StateDict",
) -> "SerializationProtocol":
    """
    Updates the object's constant (atomic) values from a partial or full dictionary.

    Automatically detects the object's operational mode:
    - Mutable mode: updates values in-place.
    - Immutable mode: returns a new instance with applied changes.
    """

    # 1. Validate dictionary input
    if not isinstance(user_dict, dict):
        raise_unvalid_state_dict_input(self, user_dict)

    # 2. Prepare list for collecting changes (immutable mode)
    changes: list[tuple[str, object]] = []

    # 3. Traverse all nodes and search for values according to their paths
    for instance, path in self._nodes_tuple:

        # 3.1 Skip composite VALUE nodes (only pure constants matter)
        if path[-1] == VALUE:
            continue

        # 3.2 Search for the value deep inside the user dictionary
        current = user_dict
        for key in path:
            current = current.get(key, UNSET)
            if not isinstance(current, dict) and key != path[-1]:
                current = UNSET
                break

        # 3.3 If a value was found → branch based on mutability
        if current is not UNSET:

            # 3.3.1 MUTABLE MODE → write directly in-place
            if self.__is_mutable__:
                instance.value = current

            # 3.3.2 IMMUTABLE MODE → collect changes for batch update
            else:
                name = instance.meta.name_snake
                changes.append((name, current))

    # 4. Handle mutable mode (and immutable mode with no changes)
    if self.__is_mutable__ or not changes:
        return self

    # 5. Handle immutable mode with collected changes
    return with_constants(self, tuple(changes))

def to_dict(
    self: "SerializationProtocol",
    *,
    constant_only: bool = False,
) -> "StateDict":
    """
    Serializes the instance into a readable dictionary structure.
    Composite nodes are expanded into nested dictionaries according to their
    structural path in `nodes_tuple`.
    """

    # 1. Define base variables
    result = {OWNER_CLASS: self.meta.owner_class}
    last_composite_key = UNSET
    last_composite_dict = {}

    # 2. Iterate through nodes_tuple in structural order
    for instance, path in self._nodes_tuple:

        # 2.1 Define per-node variables
        value = instance._value
        current_dict = result
        last_key = path[0]

        # 2.2 Handle composite nodes (non-root)
        if len(path) > 1:

            # 2.2.1 Extract last and second-to-last keys
            last_key = path[-1]
            before_last_key = path[-2]

            # 2.2.2 Reuse previously cached composite dictionary
            if last_composite_key == before_last_key:
                current_dict = last_composite_dict

            # 2.2.3 Create a new composite dictionary
            else:

                # 2.2.3.1 Build nested structure for all keys except the last
                for key in path[:-1]:
                    current_dict = current_dict.setdefault(key, {})

                # 2.2.3.2 Cache composite dictionary for next iteration
                if last_key == VALUE:
                    last_composite_key = before_last_key
                    last_composite_dict = current_dict

        # 2.3 Write the node value (unless filtered out)
        if not (constant_only and last_key == VALUE):
            current_dict[last_key] = value

    # 3. Return the final dictionary
    return result
